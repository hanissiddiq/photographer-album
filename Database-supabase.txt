-- =========================================================
-- 1. ENUM ROLE
-- =========================================================

create type public.user_role as enum (
    'photographer',
    'client'
);


-- =========================================================
-- 2. PROFILES TABLE
-- =========================================================

create table public.profiles (
    id uuid primary key references auth.users(id) on delete cascade,

    full_name text,

    email text,

    phone text,

    role public.user_role not null default 'client',

    avatar_url text,

    is_active boolean not null default true,

    created_at timestamptz not null default now(),

    updated_at timestamptz not null default now()
);

alter table public.profiles
add column if not exists whatsapp_number text;


-- =========================================================
-- 3. INDEX
-- =========================================================

create index profiles_role_idx
    on public.profiles(role);

create index profiles_email_idx
    on public.profiles(email);


-- =========================================================
-- 4. UPDATED_AT FUNCTION
-- =========================================================

create or replace function public.handle_updated_at()
returns trigger
language plpgsql
security invoker
set search_path = public
as $$
begin
    new.updated_at = now();
    return new;
end;
$$;


-- =========================================================
-- 5. UPDATED_AT TRIGGER
-- =========================================================

create trigger profiles_updated_at
before update on public.profiles
for each row
execute function public.handle_updated_at();


-- =========================================================
-- 6. CREATE PROFILE AFTER REGISTER
-- =========================================================

create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin

    insert into public.profiles (
        id,
        full_name,
        email,
        phone,
        role
    )
    values (
        new.id,
        coalesce(
            new.raw_user_meta_data ->> 'full_name',
            ''
        ),
        new.email,
        coalesce(
            new.raw_user_meta_data ->> 'phone',
            ''
        ),

        'client'::public.user_role
    );

    return new;

end;
$$;


-- =========================================================
-- 7. AUTH USER TRIGGER
-- =========================================================

create trigger on_auth_user_created
after insert on auth.users
for each row
execute function public.handle_new_user();

create type public.album_status as enum (
    'selection',
    'editing',
    'printing',
    'done'
);


create table public.albums (
    id uuid primary key default gen_random_uuid(),

    photographer_id uuid not null
        references public.profiles(id)
        on delete cascade,

    client_id uuid
        references public.profiles(id)
        on delete set null,

    title text not null,

    slug text not null unique,

    description text,

    drive_url text,

    quota integer not null default 40,

    whatsapp_number text,

    status public.album_status not null default 'selection',

    is_active boolean not null default true,

    expires_at timestamptz,

    created_at timestamptz not null default now(),

    updated_at timestamptz not null default now(),

    constraint albums_quota_positive
        check (quota > 0)
);

create index albums_photographer_id_idx
on public.albums(photographer_id);

create index albums_client_id_idx
on public.albums(client_id);

create index albums_status_idx
on public.albums(status);

create index albums_slug_idx
on public.albums(slug);

create index albums_is_active_idx
on public.albums(is_active);

create trigger albums_updated_at
before update on public.albums
for each row
execute function public.handle_updated_at();

-- =========================================================
-- Photo Selection
-- =========================================================

create table public.album_selections (
    id uuid primary key default gen_random_uuid(),

    album_id uuid not null
        references public.albums(id)
        on delete cascade,

    access_token_id uuid not null
        references public.album_access_tokens(id)
        on delete cascade,

    selected_count integer not null default 0,

    status text not null default 'draft'
        check (
            status in (
                'draft',
                'submitted',
                'editing',
                'printing',
                'done'
            )
        ),

    submitted_at timestamptz,

    created_at timestamptz not null default now(),

    updated_at timestamptz not null default now(),

    constraint album_selections_token_unique
        unique(access_token_id)
);

create table public.album_selected_photos (
    id uuid primary key default gen_random_uuid(),

    selection_id uuid not null
        references public.album_selections(id)
        on delete cascade,

    album_photo_id uuid not null
        references public.album_photos(id)
        on delete cascade,

    selected_at timestamptz not null default now(),

    constraint album_selected_photos_unique
        unique(selection_id, album_photo_id)
);

create index album_selections_album_id_idx
on public.album_selections(album_id);

create index album_selected_photos_selection_id_idx
on public.album_selected_photos(selection_id);

create index album_selected_photos_photo_id_idx
on public.album_selected_photos(album_photo_id);

alter table public.album_selections
enable row level security;

alter table public.album_selected_photos
enable row level security;

-- ===================================
-- RPC Submit Album Selection
-- ===================================

create or replace function public.submit_album_selection(
    p_selection_id uuid,
    p_album_id uuid,
    p_access_token_id uuid
)
returns jsonb
language plpgsql
security definer
set search_path = public
as $$
declare
    v_selection public.album_selections%rowtype;
    v_album public.albums%rowtype;
    v_selected_count integer;
begin

    /*
    |--------------------------------------------------------------------------
    | 1. Lock selection
    |--------------------------------------------------------------------------
    */

    select *
    into v_selection
    from public.album_selections
    where id = p_selection_id
      and album_id = p_album_id
      and access_token_id = p_access_token_id
    for update;

    if not found then
        raise exception 'Selection tidak ditemukan.';
    end if;


    /*
    |--------------------------------------------------------------------------
    | 2. Selection harus masih draft
    |--------------------------------------------------------------------------
    */

    if v_selection.status <> 'draft' then

        raise exception
            'Selection sudah dikirim dan tidak dapat diubah.';

    end if;


    /*
    |--------------------------------------------------------------------------
    | 3. Ambil album
    |--------------------------------------------------------------------------
    */

    select *
    into v_album
    from public.albums
    where id = p_album_id
      and is_active = true;

    if not found then
        raise exception 'Album tidak aktif atau tidak ditemukan.';
    end if;


    /*
    |--------------------------------------------------------------------------
    | 4. Check album expiration
    |--------------------------------------------------------------------------
    */

    if v_album.expires_at is not null
       and v_album.expires_at < now() then

        raise exception 'Album sudah kedaluwarsa.';

    end if;


    /*
    |--------------------------------------------------------------------------
    | 5. Hitung foto yang dipilih
    |--------------------------------------------------------------------------
    */

    select count(*)
    into v_selected_count
    from public.album_selected_photos
    where selection_id = v_selection.id;


    /*
    |--------------------------------------------------------------------------
    | 6. Update count agar selalu sinkron
    |--------------------------------------------------------------------------
    */

    update public.album_selections
    set
        selected_count = v_selected_count,
        updated_at = now()
    where id = v_selection.id;


    /*
    |--------------------------------------------------------------------------
    | 7. HARUS tepat sama dengan quota
    |--------------------------------------------------------------------------
    */

    if v_selected_count <> v_album.quota then

        raise exception
            'Jumlah foto yang dipilih harus tepat % foto. Saat ini % foto.',
            v_album.quota,
            v_selected_count;

    end if;


    /*
    |--------------------------------------------------------------------------
    | 8. Final submit
    |--------------------------------------------------------------------------
    */

    update public.album_selections
    set
        status = 'submitted',
        submitted_at = now(),
        selected_count = v_selected_count,
        updated_at = now()
    where id = v_selection.id;


    /*
    |--------------------------------------------------------------------------
    | 9. Return result
    |--------------------------------------------------------------------------
    */

    return jsonb_build_object(
        'success', true,
        'selection_id', v_selection.id,
        'album_id', v_album.id,
        'selected_count', v_selected_count,
        'quota', v_album.quota,
        'status', 'submitted'
    );

end;
$$;
